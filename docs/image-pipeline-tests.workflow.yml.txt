name: image-pipeline-tests

# WARP-663 / ADR-020 appliance image pipeline. Runs the unit + static checks
# that do NOT need a real ISO build or hardware (those are the manual Linux
# flash+boot gate documented in docs/IMAGE_PIPELINE.md).

on:
  # Temporarily disabled: manual-only until GitHub Actions testing minutes
  # are restored. Restore the pull_request/push triggers from git history.
  workflow_dispatch:

concurrency:
  group: image-pipeline-tests-${{ github.ref }}
  cancel-in-progress: true

jobs:
  image-pipeline:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4

      - name: Install shellcheck
        run: sudo apt-get update -qq && sudo apt-get install -y -qq shellcheck

      - name: Unit tests (droplet-image CLI, sign/verify, manifest, flash safety)
        run: bash tests/image-pipeline.test.sh

      - name: Static check (build-image non-stub, schema, manifest, shellcheck)
        run: bash scripts/test/ship-check.sh image-pipeline
