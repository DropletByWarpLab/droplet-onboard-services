# Wake-word stack (commit 4, WARP-154).
#
# openWakeWord ships an ONNX backend natively — we do NOT use tflite-runtime
# (the upstream default), which has no Python 3.12 wheel for x86_64. The
# Dockerfile installs `openwakeword` with `--no-deps` to skip the tflite-runtime
# pin in its setup.py, and we ship the ONNX-side deps explicitly here so
# `pip install -r requirements.txt` from a dev box still works.
onnxruntime==1.28.0       # ONNX inference engine (openWakeWord's --inference-framework=onnx)
scipy==1.18.0             # openWakeWord's audio preprocessor (mel-spectrogram, resampling)
huggingface_hub==1.26.0   # First-run download of bundled models (hey_jarvis, alexa, etc.)
tqdm==4.70.0              # Progress bar during model download
pyyaml==6.0.3             # openWakeWord parses model config from YAML
requests==2.34.2          # huggingface_hub's HTTP layer
# openwakeword 0.6.0's package __init__ unconditionally imports
# custom_verifier_model → sklearn.linear_model. Even though we never
# call train_custom_verifier ourselves, ANY `import openwakeword.*`
# triggers the chain. Without sklearn the runtime falls back to
# DisabledWakeWordDetector — wake detection silently broken. ~30 MB
# but unavoidable as long as we're on openwakeword < 0.7.
scikit-learn==1.9.0

# Speaker identification (per-person voiceprints, WARP-1056).
#
# Kaldi-compatible 80-dim fbank features for the bundled CAM++ speaker
# model (see the Dockerfile's speaker-model step) — the EXACT frontend
# the model was trained/exported against, so no hand-rolled mel math to
# drift. Tiny wheel (~1 MB), Apache-2.0, manylinux wheels for py3.12.
# Inference runs on the onnxruntime pinned above.
kaldi-native-fbank==1.22.3

# Vosk wake backend (default WAKE_ENGINE=vosk, WARP-154).
#
# Recognizes the configured wake phrase ("hey droplet") out of the box
# via a grammar-constrained small acoustic model — no per-phrase model
# training and no licensing fee (Apache-2.0), unlike Porcupine-style
# engines. The ~40 MB model itself is baked into the image by the
# Dockerfile, not pulled from PyPI. Ships manylinux wheels for py3.12.
vosk==0.3.45
